FUNCTION_BLOCK FB_TimedTankFilling
VAR_INPUT
    StartCommand : BOOL;
    StopCommand : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TimerDuration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    StartEdge : R_TRIG;
    FillingTimer : TON;
    start : BOOL;
    stop : BOOL;
    timer_done : BOOL;
    FillingState : INT := 0;
END_VAR

StartEdge(CLK := StartCommand);
start := StartEdge.Q;
stop := StopCommand;

IF stop THEN
    FillingState := 0;
ELSIF FillingState = 0 THEN
    IF start AND NOT LSHH THEN
        FillingState := 1;
    END_IF;
ELSIF FillingState <> 1 THEN
    FillingState := 0;
END_IF;

(* Timing begins in the scan that activates the filling step. *)
FillingTimer(
    IN := (FillingState = 1),
    PT := TimerDuration
);
timer_done := FillingTimer.Q;

IF FillingState = 1 THEN
    IF stop OR LSHH OR timer_done THEN
        FillingState := 0;

        (* Reset the step-owned timer immediately upon leaving filling. *)
        FillingTimer(IN := FALSE, PT := TimerDuration);
        timer_done := FillingTimer.Q;
    END_IF;
END_IF;

fill_valve := (FillingState = 1);
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    TankController : FB_TimedTankFilling;
    TimerDuration : TIME := T#10s; (* Configurable filling-time preset. *)
END_VAR

TankController(
    StartCommand := StartCommand,
    StopCommand := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TimerDuration := TimerDuration
);

FillValve := TankController.fill_valve;
DischargeValve := TankController.discharge_valve;

END_PROGRAM

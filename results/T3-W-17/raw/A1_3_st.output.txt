FUNCTION_BLOCK TankRuntime
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TON_Duration : TIME;
    TOF_Duration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    ActiveState : INT := 0;
    PreviousState : INT := 0;
    FillingActive : BOOL;
    FillTimer : TON;
    PauseTimer : TOF;
    ton_q : BOOL;
    tof_q : BOOL;
END_VAR

(* Evaluate both timers cyclically using filling-step activity. *)
FillingActive := (ActiveState = 1);
FillTimer(IN := FillingActive, PT := TON_Duration);
PauseTimer(IN := FillingActive, PT := TOF_Duration);
ton_q := FillTimer.Q;
tof_q := PauseTimer.Q;

PreviousState := ActiveState;

CASE ActiveState OF
    0:
        IF start AND NOT stop AND NOT LSH AND NOT LSHH THEN
            ActiveState := 1;
        END_IF;

    1:
        IF stop OR LSH OR LSHH THEN
            ActiveState := 0;
        ELSIF ton_q AND NOT stop AND NOT LSH AND NOT LSHH THEN
            ActiveState := 2;
        END_IF;

    2:
        IF stop OR LSH OR LSHH THEN
            ActiveState := 0;
        ELSIF NOT tof_q AND NOT stop AND NOT LSH AND NOT LSHH THEN
            ActiveState := 1;
        END_IF;

ELSE
    ActiveState := 0;
END_CASE;

(* Synchronize timer inputs immediately when filling activity changes.
   This starts the TOF delay on departure from the filling step and
   resets the TON before another filling phase can begin. *)
IF ActiveState <> PreviousState THEN
    FillingActive := (ActiveState = 1);
    FillTimer(IN := FillingActive, PT := TON_Duration);
    PauseTimer(IN := FillingActive, PT := TOF_Duration);
    ton_q := FillTimer.Q;
    tof_q := PauseTimer.Q;
END_IF;

fill_valve := (ActiveState = 1);
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    (* Configurable preset values; adjust to the required process timing. *)
    TON_Duration : TIME := T#5s;
    TOF_Duration : TIME := T#5s;
    StartEdge : R_TRIG;
    start : BOOL;
    Controller : TankRuntime;
END_VAR

StartEdge(CLK := StartCommand);
start := StartEdge.Q;

Controller(
    start := start,
    stop := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TON_Duration := TON_Duration,
    TOF_Duration := TOF_Duration
);

FillValve := Controller.fill_valve;
DischargeValve := Controller.discharge_valve;

END_PROGRAM

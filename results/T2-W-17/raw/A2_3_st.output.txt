FUNCTION_BLOCK FB_TimedFilling
VAR_INPUT
    StartCommand : BOOL;
    StopCommand : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TimerDuration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    StartEdge : R_TRIG;
    FillingTimer : TON;
    SequenceState : INT := 0;
    start : BOOL;
    stop : BOOL;
    timer_done : BOOL;
END_VAR

StartEdge(CLK := StartCommand);
start := StartEdge.Q;
stop := StopCommand;

IF stop OR LSHH THEN
    SequenceState := 0;
ELSE
    CASE SequenceState OF
        0:
            IF start THEN
                SequenceState := 1;
            END_IF;
        1:
            SequenceState := 1;
    ELSE
        SequenceState := 0;
    END_CASE;
END_IF;

FillingTimer(
    IN := (SequenceState = 1),
    PT := TimerDuration
);
timer_done := FillingTimer.Q;

IF (SequenceState = 1) AND timer_done THEN
    SequenceState := 0;
    FillingTimer(IN := FALSE, PT := TimerDuration);
    timer_done := FALSE;
END_IF;

fill_valve := (SequenceState = 1);
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    TimerDuration : TIME := T#10s;
    FillingController : FB_TimedFilling;
END_VAR

FillingController(
    StartCommand := StartCommand,
    StopCommand := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TimerDuration := TimerDuration
);

FillValve := FillingController.fill_valve;
DischargeValve := FillingController.discharge_valve;

END_PROGRAM

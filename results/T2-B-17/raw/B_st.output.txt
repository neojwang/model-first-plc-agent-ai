FUNCTION_BLOCK FillingTankControl
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TimerDuration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    current_step : INT := 0;
    filling_timer : TON;
    timer_done : BOOL := FALSE;
END_VAR

(* Stop has priority over starting or continuing a fill. *)
IF stop THEN
    current_step := 0;
ELSE
    CASE current_step OF
        0:
            IF start AND NOT LSHH THEN
                current_step := 1;
            END_IF;

        1:
            IF LSHH THEN
                current_step := 0;
            END_IF;

        ELSE
            current_step := 0;
    END_CASE;
END_IF;

(* Timing starts in the same scan that the filling step is entered. *)
filling_timer(
    IN := (current_step = 1),
    PT := TimerDuration
);
timer_done := filling_timer.Q;

IF (current_step = 1) AND timer_done THEN
    current_step := 0;
END_IF;

fill_valve := (current_step = 1) AND NOT stop AND NOT LSHH;
discharge_valve := FALSE;

(* Reset immediately on leaving the filling step, ready for another start. *)
IF current_step = 0 THEN
    filling_timer(IN := FALSE, PT := TimerDuration);
END_IF;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;

    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    (* Adjustable filling duration. *)
    TimerDuration : TIME := T#10s;

    start : BOOL;
    stop : BOOL;
    start_edge : R_TRIG;
    controller : FillingTankControl;
END_VAR

start_edge(CLK := StartCommand);
start := start_edge.Q;
stop := StopCommand;

controller(
    start := start,
    stop := stop,
    LSH := LSH,
    LSHH := LSHH,
    TimerDuration := TimerDuration
);

FillValve := controller.fill_valve;
DischargeValve := controller.discharge_valve;

END_PROGRAM

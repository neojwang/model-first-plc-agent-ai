FUNCTION_BLOCK FillingRuntime
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TON_Duration : TIME;
    TOF_Duration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    active_state : INT := 0;
    next_state : INT := 0;
    filling_active : BOOL := FALSE;
    ton_q : BOOL := FALSE;
    tof_q : BOOL := FALSE;
    filling_timer : TON;
    pause_timer : TOF;
END_VAR

(* Update timers for the current situation before testing transitions. *)
filling_active := active_state = 1;

filling_timer(
    IN := filling_active,
    PT := TON_Duration
);
pause_timer(
    IN := filling_active,
    PT := TOF_Duration
);

ton_q := filling_timer.Q;
tof_q := pause_timer.Q;

(* Evaluate transitions from the current situation only. *)
next_state := active_state;

CASE active_state OF
    0:
        IF start AND NOT stop AND NOT LSH AND NOT LSHH THEN
            next_state := 1;
        END_IF;

    1:
        IF stop OR LSH OR LSHH THEN
            next_state := 0;
        ELSIF ton_q AND NOT stop AND NOT LSH AND NOT LSHH THEN
            next_state := 2;
        END_IF;

    2:
        IF stop OR LSH OR LSHH THEN
            next_state := 0;
        ELSIF NOT tof_q AND NOT stop AND NOT LSH AND NOT LSHH THEN
            next_state := 1;
        END_IF;

ELSE
    next_state := 0;
END_CASE;

(* Apply timer input edges immediately when the situation changes.
   In particular, leaving filling starts the TOF off-delay now. *)
IF next_state <> active_state THEN
    active_state := next_state;
    filling_active := active_state = 1;

    filling_timer(
        IN := filling_active,
        PT := TON_Duration
    );
    pause_timer(
        IN := filling_active,
        PT := TOF_Duration
    );

    ton_q := filling_timer.Q;
    tof_q := pause_timer.Q;
END_IF;

fill_valve := active_state = 1;
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;

    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    (* Configurable timer presets. *)
    TON_Duration : TIME := T#5s;
    TOF_Duration : TIME := T#5s;

    start_edge : R_TRIG;
    start : BOOL := FALSE;
    first_scan : BOOL := TRUE;
    controller : FillingRuntime;
END_VAR

start_edge(CLK := StartCommand);

(* An already-held start button does not start filling at startup. *)
IF first_scan THEN
    start := FALSE;
    first_scan := FALSE;
ELSE
    start := start_edge.Q;
END_IF;

controller(
    start := start,
    stop := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TON_Duration := TON_Duration,
    TOF_Duration := TOF_Duration
);

FillValve := controller.fill_valve;
DischargeValve := controller.discharge_valve;

END_PROGRAM

FUNCTION_BLOCK TankFillingControl
VAR_INPUT
    StartCommand : BOOL;
    StopCommand : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TON_Duration : TIME;
    TOF_Duration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    StateNumber : INT := 0;
    FillingActive : BOOL;
    StartEdge : R_TRIG;
    FillingTimer : TON;
    PauseTimer : TOF;
    start : BOOL;
    stop : BOOL;
    ton_q : BOOL;
    tof_q : BOOL;
END_VAR

StartEdge(CLK := StartCommand);
start := StartEdge.Q;
stop := StopCommand;

FillingActive := (StateNumber = 1);

FillingTimer(
    IN := FillingActive,
    PT := TON_Duration
);
PauseTimer(
    IN := FillingActive,
    PT := TOF_Duration
);

ton_q := FillingTimer.Q;
tof_q := PauseTimer.Q;

IF stop THEN
    StateNumber := 0;
ELSE
    CASE StateNumber OF
        0:
            IF start AND NOT LSH THEN
                StateNumber := 1;
            END_IF;

        1:
            IF LSH THEN
                StateNumber := 0;
            ELSIF ton_q THEN
                StateNumber := 2;
            END_IF;

        2:
            IF LSH THEN
                StateNumber := 0;
            ELSIF NOT tof_q THEN
                StateNumber := 1;
            END_IF;

        ELSE
            StateNumber := 0;
    END_CASE;
END_IF;

fill_valve := (StateNumber = 1);
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    TON_Duration : TIME;
    TOF_Duration : TIME;
    TankController : TankFillingControl;
END_VAR

TankController(
    StartCommand := StartCommand,
    StopCommand := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TON_Duration := TON_Duration,
    TOF_Duration := TOF_Duration
);

FillValve := TankController.fill_valve;
DischargeValve := TankController.discharge_valve;

END_PROGRAM

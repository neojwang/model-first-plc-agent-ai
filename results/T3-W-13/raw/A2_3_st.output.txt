FUNCTION_BLOCK TankFillingControl
VAR_INPUT
    StartCommand : BOOL;
    StopCommand : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TON_Duration : TIME;
    TOF_Duration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    active_state : INT := 0;
    start_edge : R_TRIG;
    filling_timer : TON;
    pause_timer : TOF;
    start : BOOL;
    stop : BOOL;
    ton_q : BOOL;
    tof_q : BOOL;
    x0 : BOOL;
    x1 : BOOL;
    x2 : BOOL;
END_VAR

start_edge(CLK := StartCommand);
start := start_edge.Q;
stop := StopCommand;

(* Both timers receive the activity of the filling step. *)
x1 := (active_state = 1);
filling_timer(IN := x1, PT := TON_Duration);
pause_timer(IN := x1, PT := TOF_Duration);
ton_q := filling_timer.Q;
tof_q := pause_timer.Q;

(* Evaluate transitions from the current situation only. *)
IF stop THEN
    active_state := 0;
ELSE
    CASE active_state OF
        0:
            IF start AND NOT LSH THEN
                active_state := 1;
            END_IF;

        1:
            IF LSH THEN
                active_state := 0;
            ELSIF ton_q THEN
                active_state := 2;
            END_IF;

        2:
            IF LSH THEN
                active_state := 0;
            ELSIF NOT tof_q THEN
                active_state := 1;
            END_IF;

        ELSE
            active_state := 0;
    END_CASE;
END_IF;

x0 := (active_state = 0);
x1 := (active_state = 1);
x2 := (active_state = 2);

fill_valve := x1;
discharge_valve := FALSE;

(* LSHH is an observation input, not an additional GRAFCET transition.
   Prevention of LSHH depends on termination at LSH and plant response. *)
END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    tank_controller : TankFillingControl;
    (* Commissioning defaults; adjust to the required process timings. *)
    TON_Duration : TIME := T#5s;
    TOF_Duration : TIME := T#5s;
END_VAR

tank_controller(
    StartCommand := StartCommand,
    StopCommand := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TON_Duration := TON_Duration,
    TOF_Duration := TOF_Duration
);

FillValve := tank_controller.fill_valve;
DischargeValve := tank_controller.discharge_valve;
END_PROGRAM

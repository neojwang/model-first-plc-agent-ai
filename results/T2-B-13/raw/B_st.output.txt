FUNCTION_BLOCK FillingControl
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TimerDuration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    state_id : INT := 0;
    filling_timer : TON;
    timer_done : BOOL := FALSE;
END_VAR

IF stop THEN
    state_id := 0;
ELSE
    CASE state_id OF
        0:
            IF start AND NOT LSHH THEN
                state_id := 1;
            END_IF;

        1:
            IF LSHH THEN
                state_id := 0;
            END_IF;

    ELSE
        state_id := 0;
    END_CASE;
END_IF;

filling_timer(
    IN := (state_id = 1) AND NOT stop,
    PT := TimerDuration
);
timer_done := filling_timer.Q;

IF timer_done THEN
    state_id := 0;
END_IF;

fill_valve := (state_id = 1) AND NOT stop AND NOT LSHH;
discharge_valve := FALSE;

(* Reset the timer before accepting a new filling cycle. *)
IF state_id = 0 THEN
    filling_timer(IN := FALSE, PT := TimerDuration);
    timer_done := filling_timer.Q;
END_IF;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    TimerDuration : TIME := T#10s;
    start_detector : R_TRIG;
    controller : FillingControl;
END_VAR

start_detector(CLK := StartCommand);

controller(
    start := start_detector.Q,
    stop := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TimerDuration := TimerDuration
);

FillValve := controller.fill_valve;
DischargeValve := controller.discharge_valve;

END_PROGRAM

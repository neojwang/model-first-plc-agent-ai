FUNCTION_BLOCK FB_TankFilling
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TimerDuration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    x0 : BOOL := TRUE;
    x1 : BOOL := FALSE;
    timer_done : BOOL := FALSE;
    FillingTimer : TON;
    t1 : BOOL;
    t2 : BOOL;
END_VAR

(* Evaluate the timer and transitions for the current situation. *)
FillingTimer(IN := x1, PT := TimerDuration);
timer_done := FillingTimer.Q;

t1 := x0 AND start AND NOT stop AND NOT LSHH;
t2 := x1 AND (stop OR LSHH OR timer_done);

(* Apply the enabled transition without cascading transitions. *)
IF t1 THEN
    x0 := FALSE;
    x1 := TRUE;

    (* Start timing immediately upon activation of x1. *)
    FillingTimer(IN := TRUE, PT := TimerDuration);
    timer_done := FALSE;
ELSIF t2 THEN
    x0 := TRUE;
    x1 := FALSE;

    (* Reset the step-owned timer upon deactivation of x1. *)
    FillingTimer(IN := FALSE, PT := TimerDuration);
    timer_done := FALSE;
END_IF;

fill_valve := x1;
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    (* Configurable filling duration; set to the required process value. *)
    TimerDuration : TIME := T#10s;
    StartEdge : R_TRIG;
    TankController : FB_TankFilling;
END_VAR

StartEdge(CLK := StartCommand);

TankController(
    start := StartEdge.Q,
    stop := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TimerDuration := TimerDuration
);

FillValve := TankController.fill_valve;
DischargeValve := TankController.discharge_valve;

END_PROGRAM

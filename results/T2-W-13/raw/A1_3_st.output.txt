FUNCTION_BLOCK FillingRuntime
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TimerDuration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    x0 : BOOL := TRUE;
    x1 : BOOL := FALSE;
    timer_done : BOOL := FALSE;
    FillingTimer : TON;
    t1_enabled : BOOL;
    t2_enabled : BOOL;
END_VAR

(* Evaluate the timer for the current situation. *)
FillingTimer(IN := x1, PT := TimerDuration);
timer_done := FillingTimer.Q;

(* Evaluate both transitions before changing the situation. *)
t1_enabled := x0 AND start AND NOT stop
              AND NOT LSHH AND NOT timer_done;
t2_enabled := x1 AND (stop OR LSHH OR timer_done);

IF t1_enabled THEN
    x0 := FALSE;
    x1 := TRUE;

    (* Start timing on entry to x1. *)
    FillingTimer(IN := TRUE, PT := TimerDuration);
    timer_done := FALSE;
ELSIF t2_enabled THEN
    x1 := FALSE;
    x0 := TRUE;

    (* Reset the owned timer on exit from x1. *)
    FillingTimer(IN := FALSE, PT := TimerDuration);
    timer_done := FALSE;
END_IF;

fill_valve := x1;
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    (* Configurable timer preset. *)
    TimerDuration : TIME := T#10s;
    StartEdge : R_TRIG;
    TankRuntime : FillingRuntime;
END_VAR

StartEdge(CLK := StartCommand);

TankRuntime(
    start := StartEdge.Q,
    stop := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TimerDuration := TimerDuration
);

FillValve := TankRuntime.fill_valve;
DischargeValve := TankRuntime.discharge_valve;

END_PROGRAM

FUNCTION_BLOCK FillingTankControl
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TON_Duration : TIME;
    TOF_Duration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    phase : INT := 0;
    filling_active : BOOL := FALSE;
    ton_q : BOOL := FALSE;
    tof_q : BOOL := FALSE;
    FillLimitTimer : TON;
    FillPauseTimer : TOF;
END_VAR

(* Stop and either level switch take priority over filling. *)
IF stop OR LSH OR LSHH THEN
    phase := 0;
END_IF;

filling_active := (phase = 1);

FillLimitTimer(
    IN := filling_active,
    PT := TON_Duration
);
ton_q := FillLimitTimer.Q;

FillPauseTimer(
    IN := filling_active,
    PT := TOF_Duration
);
tof_q := FillPauseTimer.Q;

IF NOT (stop OR LSH OR LSHH) THEN
    CASE phase OF
        0:
            IF start THEN
                phase := 1;
            END_IF;

        1:
            IF ton_q THEN
                phase := 2;
            END_IF;

        2:
            IF NOT tof_q THEN
                phase := 1;
            END_IF;

        ELSE
            phase := 0;
    END_CASE;
END_IF;

fill_valve := (phase = 1) AND NOT stop AND NOT LSH AND NOT LSHH;
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;

    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    (* Adjustable timer parameters; initial defaults are five seconds. *)
    TON_Duration : TIME := T#5s;
    TOF_Duration : TIME := T#5s;

    start : BOOL := FALSE;
    stop : BOOL := FALSE;
    StartEdge : R_TRIG;
    TankControl : FillingTankControl;
END_VAR

StartEdge(CLK := StartCommand);
start := StartEdge.Q;
stop := StopCommand;

TankControl(
    start := start,
    stop := stop,
    LSH := LSH,
    LSHH := LSHH,
    TON_Duration := TON_Duration,
    TOF_Duration := TOF_Duration
);

FillValve := TankControl.fill_valve;
DischargeValve := TankControl.discharge_valve;

END_PROGRAM

FUNCTION_BLOCK LevelControl
VAR_INPUT
    StartCommand : BOOL;
    StopCommand : BOOL;
    LevelMeter : INT;
    register_scale : INT;
    h_span : REAL;
    Kp : REAL;
    Ti : REAL;
    Td : REAL;
    sample_period : REAL;
END_VAR
VAR_OUTPUT
    FillValve : INT;
    DischargeValve : INT;

    start : BOOL;
    stop : BOOL;
    settled : BOOL;
    r : INT;

    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR CONSTANT
    REGION_LOW : INT := 0;
    REGION_MIDDLE : INT := 1; (* Task region MID. *)
    REGION_HIGH : INT := 2;
END_VAR
VAR
    start_edge : R_TRIG;
    hold_timer : TON;

    control_state : INT := 0;
    h : REAL;
    h_target : REAL;
    h_sp : REAL;
    h_max : REAL;
    eps : REAL;
    e : REAL;
    previous_error : REAL := 0.0;

    proportional_term : REAL;
    integral_term : REAL := 0.0;
    candidate_integral : REAL;
    derivative_term : REAL;
    raw_output : REAL;
    output_percent : REAL;
    first_pid_scan : BOOL := TRUE;
END_VAR

start_edge(CLK := StartCommand);
start := start_edge.Q;
stop := StopCommand;

h := INT_TO_REAL(LevelMeter) * h_span / INT_TO_REAL(register_scale);
h_target := 0.8 * h_span;
h_sp := h_target;
h_max := 0.9 * h_span;
eps := 0.02 * h_span;
e := h_sp - h;

IF h < h_target THEN
    r := REGION_LOW;
ELSIF h < h_max THEN
    r := REGION_MIDDLE;
ELSE
    r := REGION_HIGH;
END_IF;

hold_timer(
    IN := (control_state = 1) AND NOT stop AND (ABS(e) <= eps),
    PT := T#10s
);
settled := hold_timer.Q;

(* Safe defaults apply in idle, on completion, and on stop. *)
FillValve := 0;
DischargeValve := 0;
fill_valve := FALSE;
discharge_valve := FALSE;

IF stop THEN
    control_state := 0;
    integral_term := 0.0;
    previous_error := e;
    first_pid_scan := TRUE;
ELSE
    CASE control_state OF
        0:
            integral_term := 0.0;
            previous_error := e;
            first_pid_scan := TRUE;

            IF start AND (r <> REGION_HIGH) THEN
                control_state := 1;
            END_IF;

        1:
            IF settled OR (r = REGION_HIGH) THEN
                control_state := 0;
                integral_term := 0.0;
                previous_error := e;
                first_pid_scan := TRUE;
            ELSE
                proportional_term := Kp * e;
                derivative_term := 0.0;

                IF NOT first_pid_scan AND (sample_period > 0.0) THEN
                    derivative_term :=
                        Kp * Td * (e - previous_error) / sample_period;
                END_IF;

                candidate_integral := integral_term;
                IF (Ti > 0.0) AND (sample_period > 0.0) THEN
                    candidate_integral :=
                        integral_term + Kp * e * sample_period / Ti;
                END_IF;

                raw_output :=
                    proportional_term + candidate_integral + derivative_term;

                (* Freeze the integral whenever the requested output is clamped. *)
                IF raw_output > 100.0 THEN
                    output_percent := 100.0;
                ELSIF raw_output < 0.0 THEN
                    output_percent := 0.0;
                ELSE
                    output_percent := raw_output;
                    integral_term := candidate_integral;
                END_IF;

                previous_error := e;
                first_pid_scan := FALSE;

                fill_valve := TRUE;
                discharge_valve := TRUE;
                FillValve := REAL_TO_INT(
                    output_percent * INT_TO_REAL(register_scale) / 100.0
                );
                DischargeValve := REAL_TO_INT(
                    0.4 * INT_TO_REAL(register_scale)
                );
            END_IF;

        ELSE
            control_state := 0;
            integral_term := 0.0;
            previous_error := e;
            first_pid_scan := TRUE;
    END_CASE;
END_IF;
END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LevelMeter AT %IW100 : INT;
    FillValve AT %QW100 : INT;
    DischargeValve AT %QW101 : INT;
END_VAR
VAR CONSTANT
    FULL_SCALE : INT := 1000;
END_VAR
VAR
    controller : LevelControl;
    h_span : REAL := 10.0;

    (* Output is in percent; error is in volts. Tune for the plant. *)
    Kp : REAL := 20.0;
    Ti : REAL := 5.0;
    Td : REAL := 0.0;

    (* Must match the configured cyclic task interval. *)
    sample_period : REAL := 0.02;
END_VAR

controller(
    StartCommand := StartCommand,
    StopCommand := StopCommand,
    LevelMeter := LevelMeter,
    register_scale := FULL_SCALE,
    h_span := h_span,
    Kp := Kp,
    Ti := Ti,
    Td := Td,
    sample_period := sample_period
);

FillValve := controller.FillValve;
DischargeValve := controller.DischargeValve;
END_PROGRAM


CONFIGURATION Config0
    RESOURCE Res0 ON PLC
        TASK ControlTask(INTERVAL := T#20ms, PRIORITY := 0);
        PROGRAM MainInstance WITH ControlTask : Main;
    END_RESOURCE
END_CONFIGURATION

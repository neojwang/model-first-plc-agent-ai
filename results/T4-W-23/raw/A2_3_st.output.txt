FUNCTION_BLOCK FB_LevelControl
VAR_INPUT
    StartCommand : BOOL;
    StopCommand : BOOL;
    LevelMeter : INT;
    Kp : REAL := 10.0;
    Ti : REAL := 10.0;
    Td : REAL := 0.0;
    h_span : REAL := 10.0;
    t_h : TIME := T#10s;
    cycle_seconds : REAL := 0.02;
END_VAR
VAR_OUTPUT
    FillValve : INT;
    DischargeValve : INT;
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR CONSTANT
    FULL_SCALE : INT := 1000;
    REGION_LOW : INT := 0;
    REGION_MID : INT := 1;
    REGION_HIGH : INT := 2;
END_VAR
VAR
    start_edge : R_TRIG;
    hold_timer : TON;
    start : BOOL;
    stop : BOOL;
    settled : BOOL;
    r : INT;
    state_id : INT := 0;
    previous_state : INT := 0;
    x0 : BOOL := TRUE;
    x1 : BOOL := FALSE;
    h : REAL;
    h_sp : REAL;
    h_target : REAL;
    h_max : REAL;
    eps : REAL;
    error_value : REAL;
    previous_error : REAL := 0.0;
    integral_value : REAL := 0.0;
    integral_candidate : REAL;
    proportional_value : REAL;
    derivative_value : REAL;
    candidate_output : REAL;
    raw_output : REAL;
    bounded_output : REAL;
END_VAR

start_edge(CLK := StartCommand);
start := start_edge.Q;
stop := StopCommand;

h := INT_TO_REAL(LevelMeter) * h_span / INT_TO_REAL(FULL_SCALE);
h_sp := 0.8 * h_span;
h_target := h_sp;
h_max := 0.9 * h_span;
eps := 0.02 * h_span;
error_value := h_sp - h;

IF h < h_target THEN
    r := REGION_LOW;
ELSIF h < h_max THEN
    r := REGION_MID;
ELSE
    r := REGION_HIGH;
END_IF;

hold_timer(
    IN := (state_id = 1) AND NOT stop
          AND (ABS(error_value) <= eps),
    PT := t_h
);
settled := hold_timer.Q;

previous_state := state_id;

IF stop THEN
    state_id := 0;
ELSE
    CASE state_id OF
        0:
            IF start AND (r <> REGION_HIGH) THEN
                state_id := 1;
            END_IF;
        1:
            IF settled OR (r = REGION_HIGH) THEN
                state_id := 0;
            END_IF;
    ELSE
        state_id := 0;
    END_CASE;
END_IF;

x0 := state_id = 0;
x1 := state_id = 1;

fill_valve := x1;
discharge_valve := x1;

FillValve := 0;
DischargeValve := 0;

IF x1 THEN
    IF previous_state <> 1 THEN
        previous_error := error_value;
    END_IF;

    proportional_value := Kp * error_value;

    derivative_value := 0.0;
    IF (cycle_seconds > 0.0) AND (Td > 0.0) THEN
        derivative_value :=
            Kp * Td * (error_value - previous_error) / cycle_seconds;
    END_IF;

    raw_output :=
        proportional_value + integral_value + derivative_value;

    (* Freeze integration whenever the current PID demand is clamped.
       Also reject an integration update that would cross either limit. *)
    IF (raw_output >= 0.0) AND (raw_output <= 100.0) THEN
        integral_candidate := integral_value;

        IF (Ti > 0.0) AND (cycle_seconds > 0.0) THEN
            integral_candidate :=
                integral_value + Kp * error_value * cycle_seconds / Ti;
        END_IF;

        candidate_output :=
            proportional_value + integral_candidate + derivative_value;

        IF (candidate_output >= 0.0) AND (candidate_output <= 100.0) THEN
            integral_value := integral_candidate;
            raw_output := candidate_output;
        END_IF;
    END_IF;

    IF raw_output < 0.0 THEN
        bounded_output := 0.0;
    ELSIF raw_output > 100.0 THEN
        bounded_output := 100.0;
    ELSE
        bounded_output := raw_output;
    END_IF;

    FillValve :=
        REAL_TO_INT(bounded_output * INT_TO_REAL(FULL_SCALE) / 100.0);
    DischargeValve :=
        REAL_TO_INT(0.4 * INT_TO_REAL(FULL_SCALE));

    previous_error := error_value;
END_IF;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LevelMeter AT %IW100 : INT;
    FillValve AT %QW100 : INT;
    DischargeValve AT %QW101 : INT;
END_VAR
VAR
    controller : FB_LevelControl;
    fill_valve : BOOL;
    discharge_valve : BOOL;
    (* Commissioning defaults; tune gains for the simulated tank.
       PID demand is in percent and level error is in volts. *)
    Kp : REAL := 10.0;
    Ti : REAL := 10.0;
    Td : REAL := 0.0;
    h_span : REAL := 10.0;
    t_h : TIME := T#10s;
    cycle_seconds : REAL := 0.02;
END_VAR

controller(
    StartCommand := StartCommand,
    StopCommand := StopCommand,
    LevelMeter := LevelMeter,
    Kp := Kp,
    Ti := Ti,
    Td := Td,
    h_span := h_span,
    t_h := t_h,
    cycle_seconds := cycle_seconds
);

FillValve := controller.FillValve;
DischargeValve := controller.DischargeValve;
fill_valve := controller.fill_valve;
discharge_valve := controller.discharge_valve;

END_PROGRAM


CONFIGURATION Config0
RESOURCE Res0 ON PLC
    TASK ControlTask(INTERVAL := T#20ms, PRIORITY := 0);
    PROGRAM MainInstance WITH ControlTask : Main;
END_RESOURCE
END_CONFIGURATION

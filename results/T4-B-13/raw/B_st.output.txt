FUNCTION_BLOCK LevelControl
VAR_INPUT
    StartCommand : BOOL;
    StopCommand : BOOL;
    LevelMeter : INT;
    FULL_SCALE : INT;
    Kp : REAL;
    Ti : REAL;
    Td : REAL;
END_VAR

VAR_OUTPUT
    FillValve : INT;
    DischargeValve : INT;

    start : BOOL;
    stop : BOOL;
    settled : BOOL;
    (* Region encoding: 0 = LOW, 1 = MID, 2 = HIGH. *)
    r : INT;
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR

VAR CONSTANT
    h_span : REAL := 10.0;
    sample_seconds : REAL := 0.02;
    t_h : TIME := T#10s;
END_VAR

VAR
    start_edge : R_TRIG;
    hold_timer : TON;
    control_state : INT := 0;

    h : REAL;
    h_sp : REAL;
    h_target : REAL;
    h_max : REAL;
    eps : REAL;
    e : REAL;

    integral_term : REAL := 0.0;
    proposed_integral : REAL;
    previous_error : REAL := 0.0;
    proportional_term : REAL;
    derivative_term : REAL;
    raw_output : REAL;
    bounded_output : REAL;
END_VAR

start_edge(CLK := StartCommand);
start := start_edge.Q;
stop := StopCommand;

h_sp := 0.8 * h_span;
h_target := h_sp;
h_max := 0.9 * h_span;
eps := 0.02 * h_span;

IF FULL_SCALE > 0 THEN
    h := INT_TO_REAL(LevelMeter) * h_span / INT_TO_REAL(FULL_SCALE);
ELSE
    h := 0.0;
END_IF;

e := h_sp - h;

IF h < h_target THEN
    r := 0; (* LOW *)
ELSIF h < h_max THEN
    r := 1; (* MID *)
ELSE
    r := 2; (* HIGH *)
END_IF;

IF stop THEN
    control_state := 0;
ELSIF (control_state = 0) AND start THEN
    control_state := 1;
    integral_term := 0.0;
    previous_error := e;
END_IF;

hold_timer(
    IN := (control_state = 1) AND NOT stop AND (ABS(e) <= eps),
    PT := t_h
);
settled := hold_timer.Q;

FillValve := 0;
DischargeValve := 0;
fill_valve := FALSE;
discharge_valve := FALSE;

CASE control_state OF
    0:
        integral_term := 0.0;
        previous_error := e;

    1:
        IF stop OR settled OR (r = 2) OR (FULL_SCALE <= 0) THEN
            control_state := 0;
            integral_term := 0.0;
            previous_error := e;
        ELSE
            proportional_term := Kp * e;
            derivative_term :=
                Kp * Td * (e - previous_error) / sample_seconds;

            IF Ti > 0.0 THEN
                proposed_integral :=
                    integral_term + Kp * e * sample_seconds / Ti;
            ELSE
                integral_term := 0.0;
                proposed_integral := 0.0;
            END_IF;

            raw_output :=
                proportional_term + proposed_integral + derivative_term;

            (* Freeze the integral whenever the proposed output is clamped. *)
            IF raw_output > 1.0 THEN
                bounded_output := 1.0;
            ELSIF raw_output < 0.0 THEN
                bounded_output := 0.0;
            ELSE
                bounded_output := raw_output;
                integral_term := proposed_integral;
            END_IF;

            previous_error := e;

            fill_valve := TRUE;
            discharge_valve := TRUE;
            FillValve :=
                REAL_TO_INT(bounded_output * INT_TO_REAL(FULL_SCALE));
            DischargeValve :=
                REAL_TO_INT(0.4 * INT_TO_REAL(FULL_SCALE));
        END_IF;

ELSE
    control_state := 0;
    integral_term := 0.0;
    previous_error := e;
END_CASE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LevelMeter AT %IW100 : INT;
    FillValve AT %QW100 : INT;
    DischargeValve AT %QW101 : INT;
END_VAR

VAR CONSTANT
    FULL_SCALE : INT := 1000;
END_VAR

VAR
    (* Initial tuning: output fraction per volt; Ti and Td are in seconds. *)
    Kp : REAL := 0.25;
    Ti : REAL := 5.0;
    Td : REAL := 0.0;

    controller : LevelControl;
END_VAR

controller(
    StartCommand := StartCommand,
    StopCommand := StopCommand,
    LevelMeter := LevelMeter,
    FULL_SCALE := FULL_SCALE,
    Kp := Kp,
    Ti := Ti,
    Td := Td
);

FillValve := controller.FillValve;
DischargeValve := controller.DischargeValve;

END_PROGRAM


CONFIGURATION Config0
    RESOURCE Res0 ON PLC
        TASK ControlTask(INTERVAL := T#20ms, PRIORITY := 0);
        PROGRAM MainInstance WITH ControlTask : Main;
    END_RESOURCE
END_CONFIGURATION

FUNCTION_BLOCK FillingRuntime
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TimerDuration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    x0 : BOOL := TRUE;
    x1 : BOOL := FALSE;
    timer_done : BOOL := FALSE;
    FillingTimer : TON;
END_VAR

(* Admit a start event only from the initial situation. *)
IF x0 AND start AND NOT stop AND NOT LSHH THEN
    x0 := FALSE;
    x1 := TRUE;
END_IF;

(* Timing begins in the same scan that x1 becomes active. *)
FillingTimer(IN := x1, PT := TimerDuration);
timer_done := FillingTimer.Q;

(* Terminate filling before updating the physical output commands. *)
IF x1 AND (stop OR LSHH OR timer_done) THEN
    x1 := FALSE;
    x0 := TRUE;

    (* Reset immediately so a subsequent start receives a fresh interval. *)
    FillingTimer(IN := FALSE, PT := TimerDuration);
    timer_done := FALSE;
END_IF;

fill_valve := x1;
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    (* Configurable filling interval. *)
    TimerDuration : TIME := T#5s;
    StartEdge : R_TRIG;
    FillingControl : FillingRuntime;
END_VAR

StartEdge(CLK := StartCommand);

FillingControl(
    start := StartEdge.Q,
    stop := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TimerDuration := TimerDuration
);

FillValve := FillingControl.fill_valve;
DischargeValve := FillingControl.discharge_valve;

END_PROGRAM

FUNCTION_BLOCK FB_LevelControl
VAR_INPUT
    StartCommand : BOOL;
    StopCommand : BOOL;
    LevelMeter : INT;
    Kp : REAL := 2.0;
    Ti : REAL := 5.0;
    Td : REAL := 0.0;
    CycleSeconds : REAL := 0.02;
    h_span : REAL := 10.0;
    t_h : TIME := T#10s;
END_VAR
VAR_OUTPUT
    FillValve : INT;
    DischargeValve : INT;
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR CONSTANT
    FULL_SCALE : INT := 1000;
END_VAR
VAR
    StartEdge : R_TRIG;
    HoldTimer : TON;

    start : BOOL;
    stop : BOOL;
    settled : BOOL;
    r : INT;

    StateNumber : INT := 0;
    x0 : BOOL := TRUE;
    x1 : BOOL := FALSE;
    PreviouslyActive : BOOL := FALSE;

    h : REAL;
    h_sp : REAL;
    h_target : REAL;
    h_max : REAL;
    eps : REAL;
    e : REAL;

    IntegralTerm : REAL := 0.0;
    CandidateIntegral : REAL;
    ProportionalTerm : REAL;
    DerivativeTerm : REAL;
    PreviousError : REAL := 0.0;
    UnboundedDemand : REAL;
    CandidateDemand : REAL;
    BoundedDemand : REAL;
END_VAR

StartEdge(CLK := StartCommand);
start := StartEdge.Q;
stop := StopCommand;

h := INT_TO_REAL(LevelMeter) * h_span / INT_TO_REAL(FULL_SCALE);
h_sp := 0.8 * h_span;
h_target := h_sp;
h_max := 0.9 * h_span;
eps := 0.02 * h_span;
e := h_sp - h;

(* Region encoding: 0 = LOW, 1 = middle region, 2 = HIGH. *)
IF h < h_target THEN
    r := 0;
ELSIF h < h_max THEN
    r := 1;
ELSE
    r := 2;
END_IF;

(* The timer measures continuous time in band during regulation. *)
HoldTimer(
    IN := (StateNumber = 1) AND NOT stop AND (ABS(e) <= eps),
    PT := t_h
);
settled := HoldTimer.Q;

IF stop THEN
    StateNumber := 0;
ELSE
    CASE StateNumber OF
        0:
            IF start THEN
                StateNumber := 1;
            END_IF;
        1:
            IF settled OR (r = 2) THEN
                StateNumber := 0;
            END_IF;
    ELSE
        StateNumber := 0;
    END_CASE;
END_IF;

(* Prevent even one scan of valve actuation on a high-level start. *)
IF r = 2 THEN
    StateNumber := 0;
END_IF;

x0 := StateNumber = 0;
x1 := StateNumber = 1;

fill_valve := x1;
discharge_valve := x1;

FillValve := 0;
DischargeValve := 0;

IF x1 THEN
    IF NOT PreviouslyActive THEN
        IntegralTerm := 0.0;
        PreviousError := e;
    END_IF;

    (* PID demand is expressed in percent of full valve opening.
       The error is expressed in level-meter engineering units. *)
    ProportionalTerm := Kp * e;
    DerivativeTerm := 0.0;

    IF CycleSeconds > 0.0 THEN
        DerivativeTerm :=
            Kp * Td * (e - PreviousError) / CycleSeconds;
    END_IF;

    UnboundedDemand :=
        ProportionalTerm + IntegralTerm + DerivativeTerm;

    CandidateIntegral := IntegralTerm;

    (* Suspend integration when clamped. Also reject an integral
       update that would move an unclamped output outside its limits. *)
    IF (UnboundedDemand >= 0.0)
        AND (UnboundedDemand <= 100.0)
        AND (Ti > 0.0)
        AND (CycleSeconds > 0.0) THEN

        CandidateIntegral :=
            IntegralTerm + Kp * e * CycleSeconds / Ti;

        CandidateDemand :=
            ProportionalTerm + CandidateIntegral + DerivativeTerm;

        IF (CandidateDemand >= 0.0)
            AND (CandidateDemand <= 100.0) THEN
            IntegralTerm := CandidateIntegral;
        END_IF;
    END_IF;

    UnboundedDemand :=
        ProportionalTerm + IntegralTerm + DerivativeTerm;

    IF UnboundedDemand > 100.0 THEN
        BoundedDemand := 100.0;
    ELSIF UnboundedDemand < 0.0 THEN
        BoundedDemand := 0.0;
    ELSE
        BoundedDemand := UnboundedDemand;
    END_IF;

    FillValve := REAL_TO_INT(
        BoundedDemand * INT_TO_REAL(FULL_SCALE) / 100.0
    );
    DischargeValve := REAL_TO_INT(
        0.4 * INT_TO_REAL(FULL_SCALE)
    );

    PreviousError := e;
ELSE
    IntegralTerm := 0.0;
    PreviousError := e;
    BoundedDemand := 0.0;
END_IF;

PreviouslyActive := x1;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LevelMeter AT %IW100 : INT;
    FillValve AT %QW100 : INT;
    DischargeValve AT %QW101 : INT;
END_VAR
VAR
    Controller : FB_LevelControl;
    Kp : REAL := 2.0;
    Ti : REAL := 5.0;
    Td : REAL := 0.0;
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR

Controller(
    StartCommand := StartCommand,
    StopCommand := StopCommand,
    LevelMeter := LevelMeter,
    Kp := Kp,
    Ti := Ti,
    Td := Td,
    CycleSeconds := 0.02,
    h_span := 10.0,
    t_h := T#10s
);

FillValve := Controller.FillValve;
DischargeValve := Controller.DischargeValve;
fill_valve := Controller.fill_valve;
discharge_valve := Controller.discharge_valve;

END_PROGRAM


CONFIGURATION Config0
    RESOURCE Res0 ON PLC
        TASK ControlTask(INTERVAL := T#20ms, PRIORITY := 0);
        PROGRAM MainInstance WITH ControlTask : Main;
    END_RESOURCE
END_CONFIGURATION

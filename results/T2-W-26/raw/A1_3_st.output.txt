FUNCTION_BLOCK FillingRuntime
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TimerDuration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    x0 : BOOL := TRUE;
    x1 : BOOL := FALSE;
    timer_done : BOOL := FALSE;
    filling_timer : TON;
    enter_filling : BOOL;
    leave_filling : BOOL;
END_VAR

(* Evaluate the timer for the currently active step. *)
filling_timer(IN := x1, PT := TimerDuration);
timer_done := filling_timer.Q;

(* Evaluate transitions from the current situation. *)
enter_filling := x0 AND start AND NOT stop AND NOT LSHH;
leave_filling := x1 AND (stop OR timer_done OR LSHH);

IF enter_filling THEN
    x0 := FALSE;
    x1 := TRUE;

    (* Start timing immediately upon activation of x1. *)
    filling_timer(IN := TRUE, PT := TimerDuration);
    timer_done := FALSE;
ELSIF leave_filling THEN
    x1 := FALSE;
    x0 := TRUE;

    (* Reset the timer when its owning step becomes inactive. *)
    filling_timer(IN := FALSE, PT := TimerDuration);
    timer_done := FALSE;
END_IF;

fill_valve := x1;
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    (* Configure this parameter for the required filling interval. *)
    TimerDuration : TIME := T#10s;
    start_edge : R_TRIG;
    filling_control : FillingRuntime;
END_VAR

start_edge(CLK := StartCommand);

filling_control(
    start := start_edge.Q,
    stop := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TimerDuration := TimerDuration
);

FillValve := filling_control.fill_valve;
DischargeValve := filling_control.discharge_valve;

END_PROGRAM

FUNCTION_BLOCK ConveyorControl
VAR_INPUT
    sensor_a : BOOL;
    sensor_b : BOOL;
END_VAR
VAR_OUTPUT
    conveyor : BOOL;
END_VAR
VAR
    conveyor_latch : RS;
END_VAR

(* Reset priority keeps the conveyor stopped while sensor B is active. *)
conveyor_latch(S := sensor_a, R1 := sensor_b);
conveyor := conveyor_latch.Q1;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    SensorA AT %IX100.0 : BOOL;
    SensorB AT %IX100.1 : BOOL;
    Conveyor AT %QX100.0 : BOOL;
END_VAR
VAR
    controller : ConveyorControl;
END_VAR

(* The bistable initializes to FALSE on a cold start. *)
controller(sensor_a := SensorA, sensor_b := SensorB);
Conveyor := controller.conveyor;

END_PROGRAM

FUNCTION_BLOCK ConveyorRuntime
VAR_INPUT
    sensor_a : BOOL;
    sensor_b : BOOL;
END_VAR
VAR_OUTPUT
    conveyor : BOOL := FALSE;
END_VAR
VAR
    command_latch : RS;
END_VAR

(* The RS instance initializes reset, representing initial step x0.
   Its set condition represents t1; its reset condition represents t2.
   Q1 represents step x1. *)
command_latch(
    S := sensor_a AND NOT sensor_b,
    R1 := sensor_b
);

conveyor := command_latch.Q1;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    SensorA AT %IX100.0 : BOOL;
    SensorB AT %IX100.1 : BOOL;
    Conveyor AT %QX100.0 : BOOL;
END_VAR
VAR
    runtime_logic : ConveyorRuntime;
    initialization_done : BOOL := FALSE;
END_VAR

IF NOT initialization_done THEN
    Conveyor := FALSE;
    initialization_done := TRUE;
END_IF;

runtime_logic(
    sensor_a := SensorA,
    sensor_b := SensorB
);

Conveyor := runtime_logic.conveyor;

END_PROGRAM

FUNCTION_BLOCK ConveyorRuntime
VAR_INPUT
    sensor_a : BOOL;
    sensor_b : BOOL;
    initialize : BOOL;
END_VAR
VAR_OUTPUT
    conveyor : BOOL;
END_VAR
VAR
    command_latch : RS;
END_VAR

IF initialize THEN
    command_latch(S := FALSE, R1 := TRUE);
END_IF;

command_latch(
    S := sensor_a AND NOT sensor_b,
    R1 := sensor_b
);

conveyor := command_latch.Q1;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    sensor_a AT %IX100.0 : BOOL;
    sensor_b AT %IX100.1 : BOOL;
    conveyor AT %QX100.0 : BOOL;
END_VAR
VAR
    runtime_logic : ConveyorRuntime;
    initialized : BOOL := FALSE;
END_VAR

runtime_logic(
    sensor_a := sensor_a,
    sensor_b := sensor_b,
    initialize := NOT initialized
);

conveyor := runtime_logic.conveyor;
initialized := TRUE;

END_PROGRAM

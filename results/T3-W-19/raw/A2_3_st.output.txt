FUNCTION_BLOCK FB_TankFilling
VAR_INPUT
    StartCommand : BOOL;
    StopCommand : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TON_Duration : TIME;
    TOF_Duration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    SequenceState : INT := 0;
    StartEdge : R_TRIG;
    FillingTON : TON;
    FillingTOF : TOF;
    start : BOOL;
    stop : BOOL;
    ton_q : BOOL;
    tof_q : BOOL;
    FillingActive : BOOL;
END_VAR

StartEdge(CLK := StartCommand);
start := StartEdge.Q;
stop := StopCommand;

(* Both standard timers measure filling-step activity. *)
FillingActive := (SequenceState = 1);
FillingTON(IN := FillingActive, PT := TON_Duration);
FillingTOF(IN := FillingActive, PT := TOF_Duration);

ton_q := FillingTON.Q;
tof_q := FillingTOF.Q;

(* Evaluate at most one sequence transition per invocation. *)
IF stop THEN
    SequenceState := 0;
ELSE
    CASE SequenceState OF
        0:
            IF start AND NOT LSH THEN
                SequenceState := 1;
            END_IF;

        1:
            IF LSH THEN
                SequenceState := 0;
            ELSIF ton_q THEN
                SequenceState := 2;
            END_IF;

        2:
            IF LSH THEN
                SequenceState := 0;
            ELSIF NOT tof_q THEN
                SequenceState := 1;
            END_IF;

        ELSE
            SequenceState := 0;
    END_CASE;
END_IF;

fill_valve := (SequenceState = 1);
discharge_valve := FALSE;

(* LSHH is monitored by the plant interface.
   The supplied GRAFCET defines no separate LSHH transition. *)
END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    TankController : FB_TankFilling;
    (* Configurable example presets; set to the required operating values. *)
    TON_Duration : TIME := T#5s;
    TOF_Duration : TIME := T#5s;
END_VAR

TankController(
    StartCommand := StartCommand,
    StopCommand := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TON_Duration := TON_Duration,
    TOF_Duration := TOF_Duration
);

FillValve := TankController.fill_valve;
DischargeValve := TankController.discharge_valve;

END_PROGRAM

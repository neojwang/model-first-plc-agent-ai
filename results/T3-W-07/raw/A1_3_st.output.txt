FUNCTION_BLOCK FB_FillingRuntime
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TON_Duration : TIME;
    TOF_Duration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    active_state : INT := 0;
    next_state : INT := 0;
    x0 : BOOL := TRUE;
    x1 : BOOL := FALSE;
    x2 : BOOL := FALSE;
    ton_q : BOOL := FALSE;
    tof_q : BOOL := FALSE;
    fill_timer : TON;
    pause_timer : TOF;
END_VAR

(* Both timers measure filling-step activity and execute every scan. *)
x0 := (active_state = 0);
x1 := (active_state = 1);
x2 := (active_state = 2);

fill_timer(IN := x1, PT := TON_Duration);
pause_timer(IN := x1, PT := TOF_Duration);

ton_q := fill_timer.Q;
tof_q := pause_timer.Q;

(* Evaluate transitions against the current situation. *)
next_state := active_state;

CASE active_state OF
    0:
        IF start AND NOT stop AND NOT LSH AND NOT LSHH THEN
            next_state := 1;
        END_IF;

    1:
        IF stop OR LSH OR LSHH THEN
            next_state := 0;
        ELSIF ton_q AND NOT stop AND NOT LSH AND NOT LSHH THEN
            next_state := 2;
        END_IF;

    2:
        IF stop OR LSH OR LSHH THEN
            next_state := 0;
        ELSIF NOT tof_q AND NOT stop AND NOT LSH AND NOT LSHH THEN
            next_state := 1;
        END_IF;

ELSE
    next_state := 0;
END_CASE;

(* Commit the new situation and evaluate continuous actions. *)
active_state := next_state;

x0 := (active_state = 0);
x1 := (active_state = 1);
x2 := (active_state = 2);

fill_valve := x1;
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    (* Configurable presets; set to the required operating durations. *)
    TON_Duration : TIME := T#5s;
    TOF_Duration : TIME := T#5s;
    start_edge : R_TRIG;
    start : BOOL := FALSE;
    stop : BOOL := FALSE;
    filling_runtime : FB_FillingRuntime;
END_VAR

start_edge(CLK := StartCommand);
start := start_edge.Q;
stop := StopCommand;

filling_runtime(
    start := start,
    stop := stop,
    LSH := LSH,
    LSHH := LSHH,
    TON_Duration := TON_Duration,
    TOF_Duration := TOF_Duration
);

FillValve := filling_runtime.fill_valve;
DischargeValve := filling_runtime.discharge_valve;

END_PROGRAM

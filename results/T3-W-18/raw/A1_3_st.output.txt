FUNCTION_BLOCK FillingRuntime
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TON_Duration : TIME;
    TOF_Duration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    active_state : INT := 0;
    previous_state : INT := 0;
    filling_active : BOOL := FALSE;
    ton_q : BOOL := FALSE;
    tof_q : BOOL := FALSE;
    fill_timer : TON;
    pause_timer : TOF;
END_VAR

(* Evaluate both timers using the current filling-step activity. *)
filling_active := (active_state = 1);

fill_timer(IN := filling_active, PT := TON_Duration);
pause_timer(IN := filling_active, PT := TOF_Duration);

ton_q := fill_timer.Q;
tof_q := pause_timer.Q;

previous_state := active_state;

(* Select at most one transition from the current situation. *)
CASE active_state OF
    0:
        IF start AND NOT stop AND NOT LSH AND NOT LSHH THEN
            active_state := 1;
        END_IF;

    1:
        IF stop OR LSH OR LSHH THEN
            active_state := 0;
        ELSIF ton_q AND NOT stop AND NOT LSH AND NOT LSHH THEN
            active_state := 2;
        END_IF;

    2:
        IF stop OR LSH OR LSHH THEN
            active_state := 0;
        ELSIF NOT tof_q AND NOT stop AND NOT LSH AND NOT LSHH THEN
            active_state := 1;
        END_IF;

ELSE
    active_state := 0;
END_CASE;

(* Synchronize timer inputs with any newly entered situation.
   In particular, leaving filling resets TON and starts the TOF delay
   in this same scan. No additional transition is evaluated here. *)
IF active_state <> previous_state THEN
    filling_active := (active_state = 1);

    fill_timer(IN := filling_active, PT := TON_Duration);
    pause_timer(IN := filling_active, PT := TOF_Duration);

    ton_q := fill_timer.Q;
    tof_q := pause_timer.Q;
END_IF;

fill_valve := (active_state = 1);
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    (* Configurable example presets; set to the required process values. *)
    TON_Duration : TIME := T#5s;
    TOF_Duration : TIME := T#5s;

    start_edge : R_TRIG;
    start : BOOL := FALSE;
    first_scan : BOOL := TRUE;
    controller : FillingRuntime;
END_VAR

(* Calling the edge detector on the first scan initializes its history.
   A button already held at startup is not treated as a new press. *)
start_edge(CLK := StartCommand);

IF first_scan THEN
    start := FALSE;
    first_scan := FALSE;
ELSE
    start := start_edge.Q;
END_IF;

controller(
    start := start,
    stop := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TON_Duration := TON_Duration,
    TOF_Duration := TOF_Duration
);

FillValve := controller.fill_valve;
DischargeValve := controller.discharge_valve;

END_PROGRAM

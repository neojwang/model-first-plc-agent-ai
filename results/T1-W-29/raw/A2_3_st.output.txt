FUNCTION_BLOCK FB_TransferControl
VAR_INPUT
    sensor_a : BOOL;
    sensor_b : BOOL;
END_VAR
VAR_OUTPUT
    conveyor : BOOL;
END_VAR
VAR
    run_latch : RS;
END_VAR

run_latch(S := sensor_a, R1 := sensor_b);
conveyor := run_latch.Q1;

END_FUNCTION_BLOCK

PROGRAM Main
VAR
    SensorA AT %IX100.0 : BOOL;
    SensorB AT %IX100.1 : BOOL;
    Conveyor AT %QX100.0 : BOOL;
END_VAR
VAR
    transfer_control : FB_TransferControl;
END_VAR

transfer_control(
    sensor_a := SensorA,
    sensor_b := SensorB
);
Conveyor := transfer_control.conveyor;

END_PROGRAM

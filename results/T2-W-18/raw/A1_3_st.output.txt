FUNCTION_BLOCK FillingRuntime
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TimerDuration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    x0 : BOOL := TRUE;
    x1 : BOOL := FALSE;
    t1 : BOOL;
    t2 : BOOL;
    timer_done : BOOL := FALSE;
    filling_timer : TON;
END_VAR

(* Update the timer for the situation active at scan entry. *)
filling_timer(IN := x1, PT := TimerDuration);
timer_done := filling_timer.Q;

(* Evaluate both transitions against the same situation. *)
t1 := x0 AND start AND NOT stop AND NOT LSHH;
t2 := x1 AND (stop OR LSHH OR timer_done);

(* Apply the enabled transition. *)
IF t1 THEN
    x0 := FALSE;
    x1 := TRUE;
ELSIF t2 THEN
    x1 := FALSE;
    x0 := TRUE;
END_IF;

(* Start timing on entry and reset timing immediately on exit. *)
IF t1 OR t2 THEN
    filling_timer(IN := x1, PT := TimerDuration);
    timer_done := FALSE;
END_IF;

(* Continuous, non-stored GRAFCET actions. *)
fill_valve := x1;
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    (* Configurable filling-time parameter; set to the required duration. *)
    TimerDuration : TIME := T#10s;
    start_edge : R_TRIG;
    filling_control : FillingRuntime;
END_VAR

start_edge(CLK := StartCommand);

filling_control(
    start := start_edge.Q,
    stop := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TimerDuration := TimerDuration
);

FillValve := filling_control.fill_valve;
DischargeValve := filling_control.discharge_valve;

END_PROGRAM

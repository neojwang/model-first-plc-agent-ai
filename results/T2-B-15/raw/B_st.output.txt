FUNCTION_BLOCK TankControl
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TimerDuration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    state_number : INT := 0;
    filling_timer : TON;
    timer_done : BOOL := FALSE;
END_VAR

fill_valve := FALSE;
discharge_valve := FALSE;

IF stop THEN
    state_number := 0;
ELSE
    CASE state_number OF
        0:
            IF start AND NOT LSHH THEN
                state_number := 1;
            END_IF;

        1:
            IF LSHH THEN
                state_number := 0;
            END_IF;

        ELSE
            state_number := 0;
    END_CASE;
END_IF;

filling_timer(
    IN := (state_number = 1) AND NOT stop,
    PT := TimerDuration
);
timer_done := filling_timer.Q;

IF timer_done THEN
    state_number := 0;
END_IF;

fill_valve := (state_number = 1) AND NOT stop AND NOT LSHH;

(* Reset immediately so a new start can begin a fresh timing cycle. *)
IF state_number = 0 THEN
    filling_timer(IN := FALSE, PT := TimerDuration);
END_IF;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    (* Configurable filling duration. *)
    TimerDuration : TIME := T#10s;
    start : BOOL := FALSE;
    stop : BOOL := FALSE;
    start_edge : R_TRIG;
    tank_controller : TankControl;
END_VAR

start_edge(CLK := StartCommand);
start := start_edge.Q;
stop := StopCommand;

tank_controller(
    start := start,
    stop := stop,
    LSH := LSH,
    LSHH := LSHH,
    TimerDuration := TimerDuration
);

FillValve := tank_controller.fill_valve;
DischargeValve := tank_controller.discharge_valve;

END_PROGRAM

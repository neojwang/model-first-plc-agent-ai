FUNCTION_BLOCK FB_TankControl
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TON_Duration : TIME;
    TOF_Duration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    state_number : INT := 0;
    filling_active : BOOL;
    ton_q : BOOL;
    tof_q : BOOL;
    fill_timer : TON;
    pause_timer : TOF;
END_VAR

(* Time the filling step. Safety inputs immediately remove its activity. *)
filling_active := (state_number = 1)
                  AND NOT stop
                  AND NOT LSH
                  AND NOT LSHH;

fill_timer(IN := filling_active, PT := TON_Duration);
ton_q := fill_timer.Q;

(* Apply safety conditions before processing normal transitions. *)
IF stop OR LSH OR LSHH THEN
    state_number := 0;
ELSE
    CASE state_number OF
        0:
            IF start THEN
                state_number := 1;
            END_IF;

        1:
            IF ton_q THEN
                state_number := 2;
            END_IF;

        2:
            (* The pause transition is evaluated after updating TOF. *)
            state_number := 2;

    ELSE
        state_number := 0;
    END_CASE;
END_IF;

(* Update TOF after departures from filling so its off-delay starts
   in the same scan that the fill valve closes. *)
filling_active := (state_number = 1)
                  AND NOT stop
                  AND NOT LSH
                  AND NOT LSHH;

pause_timer(IN := filling_active, PT := TOF_Duration);
tof_q := pause_timer.Q;

IF (state_number = 2) AND NOT tof_q
   AND NOT stop AND NOT LSH AND NOT LSHH THEN
    state_number := 1;
END_IF;

fill_valve := (state_number = 1)
              AND NOT stop
              AND NOT LSH
              AND NOT LSHH;
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;
    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    (* Configurable timing parameters; adjust to suit the plant. *)
    TON_Duration : TIME := T#5s;
    TOF_Duration : TIME := T#5s;

    start : BOOL;
    stop : BOOL;
    start_edge : R_TRIG;
    tank_control : FB_TankControl;
END_VAR

start_edge(CLK := StartCommand);
start := start_edge.Q;
stop := StopCommand;

tank_control(
    start := start,
    stop := stop,
    LSH := LSH,
    LSHH := LSHH,
    TON_Duration := TON_Duration,
    TOF_Duration := TOF_Duration
);

FillValve := tank_control.fill_valve;
DischargeValve := tank_control.discharge_valve;

END_PROGRAM

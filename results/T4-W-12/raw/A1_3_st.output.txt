FUNCTION_BLOCK LevelControlRuntime
VAR_INPUT
    StartCommand : BOOL;
    StopCommand : BOOL;
    MeasuredLevel : REAL;
    h_sp : REAL;
    h_target : REAL;
    h_max : REAL;
    eps : REAL;
    t_h : TIME;
    Kp : REAL;
    Ti : REAL;
    Td : REAL;
    CycleSeconds : REAL;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL := FALSE;
    discharge_valve : BOOL := FALSE;
    PIDPercent : REAL := 0.0;
END_VAR
VAR
    StartEdge : R_TRIG;
    HoldTimer : TON;

    start : BOOL;
    stop : BOOL;
    settled : BOOL := FALSE;

    (* Region encoding: 0 = LOW, 1 = MID, 2 = HIGH. *)
    r : INT := 0;

    x0 : BOOL := TRUE;
    x1 : BOOL := FALSE;
    WasRegulating : BOOL := FALSE;

    LevelError : REAL;
    PreviousError : REAL := 0.0;
    ProportionalTerm : REAL := 0.0;
    IntegralTerm : REAL := 0.0;
    DerivativeTerm : REAL := 0.0;
    CandidateIntegral : REAL := 0.0;
    CandidateDemand : REAL := 0.0;
    UnclampedDemand : REAL := 0.0;
END_VAR

StartEdge(CLK := StartCommand);
start := StartEdge.Q;
stop := StopCommand;

LevelError := h_sp - MeasuredLevel;

IF MeasuredLevel < h_target THEN
    r := 0;
ELSIF MeasuredLevel < h_max THEN
    r := 1;
ELSE
    r := 2;
END_IF;

(* Only uninterrupted time inside the tolerance band during x1 counts.
   The timer is reset during x0, so settled is FALSE on entry to x1. *)
HoldTimer(
    IN := x1 AND (ABS(LevelError) <= eps) AND NOT stop,
    PT := t_h
);
settled := HoldTimer.Q;

(* Evaluate transitions from the situation at the start of this scan.
   Only one transition can fire in a scan. *)
IF x0 THEN
    IF start AND NOT stop AND NOT (r = 2) THEN
        x0 := FALSE;
        x1 := TRUE;
    END_IF;
ELSIF x1 THEN
    IF stop OR settled OR (r = 2) THEN
        x1 := FALSE;
        x0 := TRUE;
    END_IF;
ELSE
    x0 := TRUE;
    x1 := FALSE;
END_IF;

fill_valve := x1;
discharge_valve := x1;

IF x1 THEN
    (* Reset PID state on each new regulation run.
       Initializing the error history prevents an entry derivative kick. *)
    IF NOT WasRegulating THEN
        IntegralTerm := 0.0;
        PreviousError := LevelError;
    END_IF;

    ProportionalTerm := Kp * LevelError;
    DerivativeTerm := 0.0;

    IF (Td > 0.0) AND (CycleSeconds > 0.0) THEN
        DerivativeTerm :=
            Kp * Td * (LevelError - PreviousError) / CycleSeconds;
    END_IF;

    IF Ti <= 0.0 THEN
        IntegralTerm := 0.0;
    END_IF;

    UnclampedDemand :=
        ProportionalTerm + IntegralTerm + DerivativeTerm;

    (* Conditional integration: freeze the integral when the present
       demand is outside the limits or the proposed update would
       put it outside the limits. *)
    IF (Ti > 0.0) AND (CycleSeconds > 0.0) THEN
        IF (UnclampedDemand >= 0.0)
            AND (UnclampedDemand <= 100.0) THEN
            CandidateIntegral :=
                IntegralTerm + Kp * LevelError * CycleSeconds / Ti;
            CandidateDemand :=
                ProportionalTerm + CandidateIntegral + DerivativeTerm;

            IF (CandidateDemand >= 0.0)
                AND (CandidateDemand <= 100.0) THEN
                IntegralTerm := CandidateIntegral;
            END_IF;
        END_IF;
    END_IF;

    UnclampedDemand :=
        ProportionalTerm + IntegralTerm + DerivativeTerm;

    IF UnclampedDemand > 100.0 THEN
        PIDPercent := 100.0;
    ELSIF UnclampedDemand < 0.0 THEN
        PIDPercent := 0.0;
    ELSE
        PIDPercent := UnclampedDemand;
    END_IF;

    PreviousError := LevelError;
ELSE
    PIDPercent := 0.0;
    IntegralTerm := 0.0;
    ProportionalTerm := 0.0;
    DerivativeTerm := 0.0;
    PreviousError := LevelError;
END_IF;

WasRegulating := x1;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LevelMeter AT %IW100 : INT;
    FillValve AT %QW100 : INT;
    DischargeValve AT %QW101 : INT;
END_VAR
VAR CONSTANT
    FULL_SCALE : INT := 1000;
    CycleSeconds : REAL := 0.02;
END_VAR
VAR
    Controller : LevelControlRuntime;

    h_span : REAL := 10.0;
    h_sp : REAL;
    h_target : REAL;
    h_max : REAL;
    eps : REAL;
    t_h : TIME := T#10s;

    (* Commissioning defaults; tune for the plant.
       Kp: percent opening per level volt; Ti and Td: seconds. *)
    Kp : REAL := 25.0;
    Ti : REAL := 5.0;
    Td : REAL := 0.0;

    MeasuredLevel : REAL;
END_VAR

h_sp := 0.8 * h_span;
h_target := h_sp;
h_max := 0.9 * h_span;
eps := 0.02 * h_span;

MeasuredLevel :=
    INT_TO_REAL(LevelMeter) * h_span / INT_TO_REAL(FULL_SCALE);

Controller(
    StartCommand := StartCommand,
    StopCommand := StopCommand,
    MeasuredLevel := MeasuredLevel,
    h_sp := h_sp,
    h_target := h_target,
    h_max := h_max,
    eps := eps,
    t_h := t_h,
    Kp := Kp,
    Ti := Ti,
    Td := Td,
    CycleSeconds := CycleSeconds
);

(* Explicit zero defaults prevent retained analog commands in step 0. *)
FillValve := 0;
DischargeValve := 0;

IF Controller.fill_valve THEN
    FillValve := REAL_TO_INT(
        Controller.PIDPercent * INT_TO_REAL(FULL_SCALE) / 100.0
    );
END_IF;

IF Controller.discharge_valve THEN
    DischargeValve := REAL_TO_INT(0.4 * INT_TO_REAL(FULL_SCALE));
END_IF;

END_PROGRAM


CONFIGURATION Config0
    RESOURCE Res0 ON PLC
        TASK MainTask(INTERVAL := T#20ms, PRIORITY := 0);
        PROGRAM MainInstance WITH MainTask : Main;
    END_RESOURCE
END_CONFIGURATION

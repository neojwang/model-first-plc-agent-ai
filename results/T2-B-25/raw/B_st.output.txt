FUNCTION_BLOCK TankFillingRuntime
VAR_INPUT
    start : BOOL;
    stop : BOOL;
    LSH : BOOL;
    LSHH : BOOL;
    TimerDuration : TIME;
END_VAR
VAR_OUTPUT
    fill_valve : BOOL;
    discharge_valve : BOOL;
END_VAR
VAR
    control_state : INT := 0;
    filling_timer : TON;
    timer_done : BOOL := FALSE;
END_VAR

(* Stop has priority over all state transitions. *)
IF stop THEN
    control_state := 0;
ELSE
    CASE control_state OF
        0:
            IF start AND NOT LSHH THEN
                control_state := 1;
            END_IF;

        1:
            IF LSHH THEN
                control_state := 0;
            END_IF;

        ELSE
            control_state := 0;
    END_CASE;
END_IF;

(* Start timing in the same scan that filling starts. *)
filling_timer(
    IN := (control_state = 1) AND NOT stop AND NOT LSHH,
    PT := TimerDuration
);
timer_done := filling_timer.Q;

IF (control_state = 1) AND timer_done THEN
    control_state := 0;
END_IF;

fill_valve := (control_state = 1) AND NOT stop AND NOT LSHH;
discharge_valve := FALSE;

END_FUNCTION_BLOCK


PROGRAM Main
VAR
    StartCommand AT %IX100.0 : BOOL;
    StopCommand AT %IX100.1 : BOOL;
    LSH AT %IX100.2 : BOOL;
    LSHH AT %IX100.3 : BOOL;

    FillValve AT %QX100.0 : BOOL;
    DischargeValve AT %QX100.1 : BOOL;
END_VAR
VAR
    TimerDuration : TIME := T#10s;
    start_edge : R_TRIG;
    tank_control : TankFillingRuntime;
END_VAR

(* Convert the start button into a one-scan event. *)
start_edge(CLK := StartCommand);

tank_control(
    start := start_edge.Q,
    stop := StopCommand,
    LSH := LSH,
    LSHH := LSHH,
    TimerDuration := TimerDuration
);

FillValve := tank_control.fill_valve;
DischargeValve := tank_control.discharge_valve;

END_PROGRAM
